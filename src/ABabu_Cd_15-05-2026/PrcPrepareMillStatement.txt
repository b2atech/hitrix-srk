	   dbo.tblPurch.PurBillAmt,tblMastAccount_Pty.AcBST,tblMastAccount_Pty.AcCST,        
	   dbo.tblMastItem.ItName,dbo.tblPurchSub.PurSubBag,dbo.tblPurchSub.PurSubWt,dbo.tblPurchSub.PurSubItSrNo, tblMastNarration_Trnpt.Narration, @UCODE      
from   dbo.tblPurch inner join  dbo.tblMastAccount tblMastAccount_Pty on dbo.tblPurch.PurAcCrCode = tblMastAccount_Pty.AcCode
                   inner join  dbo.tblMastCompGroup on tblMastAccount_Pty.AcCmpCode = dbo.tblMastCompGroup.CGCode   
       inner join dbo.tblMastAccount  tblMastAccount_Br on dbo.tblPurch.PurBroker = tblMastAccount_Br.AcCode
	   inner join dbo.tblMastCompany on dbo.tblPurch.VFirm = dbo.tblMastCompany.CCode   
	   inner join dbo.tblMastAccount  tblMastAccount_Ml on dbo.tblPurch.PurMillCode  = tblMastAccount_Ml.AcCode 
	   inner join dbo.tblPurchSub  on dbo.tblPurch.VNo = dbo.tblPurchSub.Vno and dbo.tblPurch.VType  = dbo.tblPurchSub.VType and dbo.tblPurch.VYear = dbo.tblPurchSub.VYear and dbo.tblPurch.VFirm  = dbo.tblPurchSub.VFirm 
	   inner join dbo.tblMastItem  on dbo.tblPurchSub.PurSubItCode  = dbo.tblMastItem.ItCode 
	   inner join dbo.tblMastNarration   on dbo.tblPurch.PurTaxCode  = dbo.tblMastNarration.NarrCode  
	   inner join dbo.tblMastNarration tblMastNarration_Trnpt  on dbo.tblPurch.PurTransport  = tblMastNarration_Trnpt.NarrCode  

where  dbo.tblPurch.VDt >= @Fr_dt and  dbo.tblPurch.VDt <= @To_dt and 
       (isnull(@VFirm,'')='' or dbo.tblPurch.vfirm = @VFirm) and
	   (isnull(@MillCode,'')='' or dbo.tblPurch.PurAcCrCode  in ( @MillCode )) and
	   (dbo.tblPurch.VType) = 'PI'
end
